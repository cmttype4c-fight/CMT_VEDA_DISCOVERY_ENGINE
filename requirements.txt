# Global scheduler timezone data (FINAL CORRECTION: SOURCE EXPANSION +
# 6:00 AM IST SCHEDULER). app/worker/ist_scheduler.py uses the stdlib
# `zoneinfo` module, which needs an IANA timezone database available at
# runtime. Some minimal Docker base images (including python:3.11-slim,
# used by this repo's Dockerfile) do not reliably ship one under
# /usr/share/zoneinfo. This pure-data package is the standard,
# CPython-recommended fallback: zoneinfo uses it automatically whenever
# the OS database is missing/incomplete, so pinning it removes any
# dependency on what happens to be present in a given base image or VPS.
# (This build sandbox's own OS already has tzdata, which is how
# `zoneinfo.ZoneInfo("Asia/Kolkata")` and every app/worker/ist_scheduler.py
# test case below were verified to run correctly here -- but that does
# NOT guarantee the deployed Docker image does; this pin is what makes
# the deployed image guaranteed-correct regardless.)
tzdata==2024.2
